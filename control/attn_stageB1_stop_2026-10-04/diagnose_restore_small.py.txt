def diffs(a,b,path='$'):
    if type(a)!=type(b):return [{'path':path,'reference':a,'restored':b,'kind':'type'}]
    if isinstance(a,dict):
        result=[]
        for k in sorted(a.keys()|b.keys()):
            if k not in a or k not in b:
                result.append({'path':f'{path}.{k}','reference':a.get(k),'restored':b.get(k),'kind':'key'})
            else:result.extend(diffs(a[k],b[k],f'{path}.{k}'))
        return result
    if isinstance(a,list):
        if len(a)!=len(b):return [{'path':path,'reference':a,'restored':b,'kind':'length'}]
        return list(itertools.chain.from_iterable(diffs(x,y,f'{path}[{i}]') for i,(x,y) in enumerate(zip(a,b))))
    return [] if a==b else [{'path':path,'reference':a,'restored':b,'kind':'value'}]
