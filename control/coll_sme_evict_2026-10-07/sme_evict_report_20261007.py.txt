def publish(src,dest=None):

def mb(v):return f'{v/1e9:.3f}'
def machine_summary(name):
 folder=EV/name
 samples=0;free=[];swap=[];bad=0
 with gzip.open(folder/'health.jsonl.gz','rt') as stream:
  for line in stream:
   row=json.loads(line);samples+=1
   disk=row['disk'].splitlines()[-1].split();free.append(int(disk[3])*1024)
   swap.append(float(re.search(r'used = ([\d.]+)M',row['swap'])[1]))
   speed=re.search(r'CPU_Speed_Limit\s*=\s*(\d+)',row['thermal'])
   if ('No thermal warning level has been recorded' not in row['thermal'] or
       'No performance warning level has been recorded' not in row['thermal'] or
       (speed is not None and int(speed[1])!=100)):bad+=1
 assert samples and not bad
 return dict(run=name,samples=samples,min_free_gib=min(free)/2**30,
  swap_min_mb=min(swap),swap_max_mb=max(swap),swap_delta_mb=swap[-1]-swap[0],warnings=bad,
  cpu_pause_seconds=read(folder/'resource.json').get('cpu_pause_seconds'),
  note='RSSと機械は標本。CPU停止欄なしの初期版では外部監視を別に記録。')
