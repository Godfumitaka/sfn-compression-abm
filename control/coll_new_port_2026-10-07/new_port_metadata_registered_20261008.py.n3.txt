"""再判定の受付後にもCPUと期限を確認する。模型の走行は起動しない。"""
from datetime import datetime, timezone
from pathlib import Path
import json
import os
import sys
import time
import new_port_cpu_20261008 as C
root=Path(__file__).resolve().parent
name=sys.argv[1]
target=root/'evidence/new-port-20261007'/('n3-'+name+'-comparison-process.json')
assert not target.exists(), '既に受付から始めた再判定を重複しない'
assert not (target.parent/f'gate-{name}-metadata-checked.json').exists()
target.write_text(json.dumps({'pid':os.getpid(),'waiting_at':datetime.now().astimezone().isoformat(),'argv':sys.argv},ensure_ascii=False,indent=2)+'\n')
while True:
    if datetime.now(timezone.utc)>=datetime(2026,10,9,tzinfo=timezone.utc):
        raise RuntimeError('継続確認の期限後に新しい再判定を始めない')
    row=C.snapshot()
    if not row['thermal_ok']:raise RuntimeError('熱又は性能の警告。新しい再判定を始めない')
    if row['count']+1<=8:
        row['planned_analysis_processes']=1
        p=target.parent/f'n3-{name}-comparison-cpu-ready.json';assert not p.exists()
        p.write_text(json.dumps(row,ensure_ascii=False,indent=2)+'\n');break
    time.sleep(15)
os.execv(sys.executable,[sys.executable,str(root/'new_port_metadata_gate_20261008.py'),*sys.argv[1:]])
