+def fingerprints(root):
+    result = {}
+    for path in root.rglob("*"):
+        if not path.is_file():
+            continue
+        digest = hashlib.sha256()
+        with path.open("rb") as stream:
+            for chunk in iter(lambda: stream.read(1024 * 1024), b""):
+                digest.update(chunk)
+        result[path.relative_to(root).as_posix()] = digest.hexdigest()
+    return result
