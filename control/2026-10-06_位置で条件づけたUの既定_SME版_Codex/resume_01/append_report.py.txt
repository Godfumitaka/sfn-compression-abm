def sha(path):
    h=hashlib.sha256()
    with path.open('rb') as f:
        while block:=f.read(1024*1024):h.update(block)
    return h.hexdigest()


def table(rows,cols):
    return '\n'.join(['| '+' | '.join(label for key,label in cols)+' |',
                      '| '+' | '.join('---' for _ in cols)+' |']+
                     ['| '+' | '.join(str(row.get(key,'')) for key,label in cols)+' |' for row in rows])

        for row in json.loads((ROOT/'cases.json').read_text()):
            p=Path(row['path'])
            if p not in cases:cases.append(p)
    for case in cases:
        cmd=json.loads((case/'native_command.json').read_text());seed=int(cmd[cmd.index('--seeds')+1]);assert seed in (1,2,3)
        dest=DEST/'executions'/case.relative_to(ROOT);dest.mkdir(parents=True,exist_ok=True)
        for name in ['native_command.json','execution.json','environment_worker.json']:
            p=case/name
            if p.exists():
                shutil.copyfile(p,dest/name);provenance.append({'kind':'execution','path':str(p),'sha256':sha(p)})
        for p in case.glob('*cpu*.jsonl'):shutil.copyfile(p,dest/p.name)
        for area in ['output/side','output/ledgers','analysis/side']:

                    provenance.append({'kind':'record','path':str(p),'sha256':sha(p)})
        for name in ['tie_state.jsonl','saved_matcher.jsonl.gz','output/manifest.jsonl','output/flag.json']:
            p=case/name
            if p.exists():provenance.append({'kind':'record','path':str(p),'sha256':sha(p)})
    with (DEST/'paths_sha256.csv').open('w') as f:
        w=csv.DictWriter(f,fieldnames=['kind','path','sha256']);w.writeheader();w.writerows(provenance)
    status=json.loads((ROOT/'pipeline_status.json').read_text())
    done=status['stage']=='pipeline' and status['status']=='completed'
    stopped=status['status']=='stopped'

    links=STEM+'/resume_01'
    rows=[]
    for mode in ('a','logp'):

状態：**{outcome}**。前の停止記録を保存したまま、利用者の指定で再開した。模型の位置Uの処理・照合器・分類の決まりは変更していない。今回の出力先は `codex_uposition_sme_2026-10-06/resume_01/`。種1〜3だけを対象とし、種21〜40の走行・記録解析はしていない。結果の良し悪しは書かない。

### 基準の環境と再実行の環境

基準は `codex_sme_evict_2026-10-06/evict_full_01/A_evict` と `L_evict` の `command.json`、`run_spec.json`、`model_pid.json` を読み、起動した台本 `run_case_01.py` まで辿った。台本の28〜30行が `PYTHONHASHSEED=0` を設定し、`LC_ALL`・`LANG`・`LC_CTYPE` を外している。前の位置Uの関門の入口はこの設定をしていなかった。

今回もPythonの起動前に `/usr/bin/env -u LC_ALL -u LANG -u LC_CTYPE PYTHONHASHSEED=0` を付ける。基準の台本はその他のLC_*を親から継承するが、その当時の値は記録に無い。今回の親にその他のLC_*は無い。実際の親と模型のworkerの環境は `execution.json` と `environment_worker.json` に保存した。Python起動時のロケール補正によってLC_CTYPEが設定された場合は、子の実記録にその値を残す。

基準のPython指定は `/opt/homebrew/opt/python@3.12/bin/python3.12`。今回も同じ指定を使い、現在の実体は `{env['python']['resolved']}`。現在の版とビルドは `{env['python']['version']}`、機械は `{env['python']['architecture']}`。optのリンクと実体のctime・mtimeは基準の開始より前であり、同じ実体・ビルドを使うとの判断はこの証拠に基づく。基準時のsys.version文字列そのものは保存されていない。この点を、基準に版の文字列まで記録されていたという主張とは区別する。

[基準の環境・実行ファイルの指紋と日時]({links}/environment/reference_environment.json)、[基準の起動台本]({links}/environment/reference_run_case_01.py.txt)。command・specと起動PIDの記録も同じフォルダへ写した。元の材料は変更していない。

'''
    if stopped:

        for mode in ('a','logp'):

    with (CONTROL/(STEM+'.md')).open('a') as f:f.write(text)
